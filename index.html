import gradio as gr import pandas as pd import matplotlib.pyplot as plt # 1. โหลดข้อมูลจากไฟล์ steam.csv df = pd.read_csv('steam.csv') # ดึงรายชื่อเกมทั้งหมดมาทำเป็นตัวเลือกใน Dropdown (จำกัด 2,000 เกมแรก) all_game_names = df['name'].dropna().unique().tolist()[:2000] # 2. ฟังก์ชันสร้างกราฟวิเคราะห์ตลาดที่ 1: แนวเกมยอดนิยม (Top Genres) def generate_genre_chart(): plt.figure(figsize=(9, 4.5)) top_genres = df['genres'].str.split(';').explode().value_counts().head(5) ax = top_genres.plot(kind='barh', color='#66c0f4', edgecolor='#2a475e', linewidth=1.5) plt.title('Top 5 Game Genres Market Share on Steam', fontsize=13, fontweight='bold', color='white', pad=15) plt.xlabel('Number of Games', fontsize=11, color='#ffffff') plt.ylabel('Genre', fontsize=11, color='#ffffff') plt.gca().set_facecolor('#1b2838') plt.gcf().set_facecolor('#171a21') plt.tick_params(colors='white', labelsize=10) plt.gca().invert_yaxis() for spine in plt.gca().spines.values(): spine.set_color('#38607d') plt.tight_layout() path1 = 'genre_chart.png' plt.savefig(path1, format='png', facecolor='#171a21') plt.close() return path1 # 3. ฟังก์ชันสร้างกราฟวิเคราะห์ตลาดที่ 2: ความสัมพันธ์ราคาและรีวิว (Price vs Positive Ratings) def generate_scatter_chart(): plt.figure(figsize=(9, 4.5)) sample_df = df[df['price'] < 60].sample(n=1000, random_state=42) plt.scatter(sample_df['price'], sample_df['positive_ratings'], color='#a4d007', alpha=0.6, edgecolors='none', s=25) plt.title('Game Price vs Positive Ratings Distribution', fontsize=13, fontweight='bold', color='white', pad=15) plt.xlabel('Price ($)', fontsize=11, color='#ffffff') plt.ylabel('Positive Ratings', fontsize=11, color='#ffffff') plt.gca().set_facecolor('#1b2838') plt.gcf().set_facecolor('#171a21') plt.tick_params(colors='white', labelsize=10) for spine in plt.gca().spines.values(): spine.set_color('#38607d') plt.tight_layout() path2 = 'scatter_chart.png' plt.savefig(path2, format='png', facecolor='#171a21') plt.close() return path2 # 4. ฟังก์ชันประมวลผลข้อมูลและสร้าง HTML Table (ปรับหัวตารางให้ตัวหนังสือเป็นสีขาวและฟ้าชัดเจน) def explore_steam_market(selected_game): if not selected_game: filtered = df.head(15) else: filtered = df[df['name'] == selected_game] if filtered.empty: filtered = df[df['name'].str.contains(selected_game, case=False, na=False)].head(15) total_found = len(filtered) avg_price = filtered['price'].mean() if total_found > 0 else 0 avg_positive = filtered['positive_ratings'].mean() if total_found > 0 else 0 # สร้าง HTML สำหรับ KPI Cards สถิติ kpi_html = f"""

🎮 เกมที่พบในระบบ

{total_found:,} รายการ

💰 ราคาเฉลี่ย

${avg_price:.2f}

⭐ คะแนนรีวิวบวกเฉลี่ย

{avg_positive:,.0f} รีวิว

""" # สร้าง HTML Table พร้อมกำหนดสีหัวตารางให้เป็นสีขาวและฟ้าสว่างชัดเจน table_html = """
""" if total_found == 0: table_html += '' else: for idx, row in filtered.iterrows(): bg_color = "#f8fafc" if idx % 2 == 0 else "#ffffff" table_html += f""" """ table_html += """
ชื่อเกม (Name) ผู้พัฒนา (Developer) แนวเกม (Genres) ราคา ($) รีวิวเชิงบวก เวลาเล่นเฉลี่ย (นาที)
ไม่พบข้อมูลเกมที่ค้นหา
{row.get('name', '')} {row.get('developer', '')} {row.get('genres', '')} ${row.get('price', 0)} {int(row.get('positive_ratings', 0)):,} {row.get('average_playtime', 0)}
""" return kpi_html, table_html # 5. CSS จัดแต่งภาพรวมเว็บไซต์และหน้าล็อกอิน custom_css = """ body, .gradio-container { background-color: #1b2838 !important; color: #ffffff !important; } h1, h2, h3, .prose h1, .prose h2, .prose h3 { color: #ffffff !important; font-weight: bold !important; } p, span, label, .prose p, .prose span { color: #c7d5e0 !important; } /* แต่งหน้าต่าง Login ให้เป็นธีม Steam สีดำ */ .gradio-container .auth-container, div[data-testid="login"] { background-color: #1b2838 !important; border: 1px solid #38607d !important; border-radius: 12px !important; } div[data-testid="login"] h2, div[data-testid="login"] label { color: #ffffff !important; } div[data-testid="login"] input { background-color: #171a21 !important; border: 1px solid #38607d !important; color: #ffffff !important; } div[data-testid="login"] button { background: linear-gradient(to bottom, #66c0f4 0%, #417a9b 100%) !important; color: #ffffff !important; font-weight: bold !important; } """ # 6. ออกแบบหน้าจอ Dashboard ด้วย Gradio Blocks with gr.Blocks(css=custom_css) as demo: gr.Markdown("# 🎮 Steam Game Explorer & Dashboard") gr.Markdown("ระบบสำรวจและวิเคราะห์ข้อมูลตลาดเกมบนแพลตฟอร์ม Steam ระดับมืออาชีพ (วิเคราะห์จากชุดข้อมูลจริง 27,075 เกม)") with gr.Tabs(): # แท็บที่ 1: ค้นหาและสำรวจข้อมูลเกม with gr.TabItem("🔍 ค้นหาและสำรวจข้อมูลเกม"): with gr.Row(): game_dropdown = gr.Dropdown( choices=all_game_names, label="🎯 เลือกหรือพิมพ์ชื่อเกมที่ต้องการสำรวจเชิงลึก", allow_custom_value=True, scale=4 ) clear_btn = gr.Button("🔄 รีเซ็ตค่า", scale=1) market_kpi = gr.HTML() table_output = gr.HTML() # แท็บที่ 2: กราฟวิเคราะห์ภาพรวมตลาด with gr.TabItem("📈 กราฟวิเคราะห์ตลาดเกม"): gr.Markdown("### 📊 รายงานสถิติและแนวโน้มตลาดเกมบนแพลตฟอร์ม Steam") with gr.Row(): chart_1 = gr.Image(value=generate_genre_chart(), label="Top 5 Game Genres Market Share", interactive=False) chart_2 = gr.Image(value=generate_scatter_chart(), label="Price vs Positive Ratings Distribution", interactive=False) # ผูกฟังก์ชัน game_dropdown.change(fn=explore_steam_market, inputs=game_dropdown, outputs=[market_kpi, table_output]) clear_btn.click(fn=lambda: (None, *explore_steam_market("")), outputs=[game_dropdown, market_kpi, table_output]) demo.load(fn=lambda: explore_steam_market(""), outputs=[market_kpi, table_output]) # 7. รันระบบพร้อมระบบ Login และสร้าง Public URL demo.launch( share=True, auth=("admin", "999") # Username: admin / Password: 999 )
